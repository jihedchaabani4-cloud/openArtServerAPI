## 5. `/ready` Implementation with Cache and Config Timeout (VERIFIED)
Readiness handler in `routes/ready.js`:
```javascript
const router = express.Router();
let cachedStatus = null;

function getReadinessTimeoutMs() {
    return parseInt(process.env.READINESS_TIMEOUT_MS, 10) || 1500;
}

function getReadinessCacheTtlMs() {
    return parseInt(process.env.READINESS_CACHE_TTL_MS, 10) || 1500;
}

router.get("/", async (req, res) => {
    try {
        const now = Date.now();
        const cacheTtlMs = getReadinessCacheTtlMs();
        if (cachedStatus && (now - cachedStatus.timestamp) < cacheTtlMs) {
            return res.status(cachedStatus.ready ? 200 : 503).json({ ready: cachedStatus.ready });
        }
        // ... evaluate connections and pings with getReadinessTimeoutMs() ...
        cachedStatus = { ready: isReady, timestamp: Date.now() };
        return res.status(isReady ? 200 : 503).json({ ready: isReady });
    } catch (err) {
        cachedStatus = { ready: false, timestamp: Date.now() };
        return res.status(503).json({ ready: false });
    }
});
```
- **Timeout**: Enforced with `Promise.race` using `READINESS_TIMEOUT_MS` (default 1500ms; proven by Test b8).
- **Cache**: In-memory cache with `READINESS_CACHE_TTL_MS` (default 1500ms; proven by Test b7 where consecutive calls within TTL avoid Redis pings).
